  estado: "en_curso" | "completado" | "incompleto";

  porcentaje: number;

};

  sesionId: string;
  estado: "completado" | "incompleto";
  motivoFinalizacion: string | null;
  preguntasRespondidas: number;
  aciertos: number;
  porcentaje: number;
  conteos: Record<Competencia, number>;
  coberturaCompleta: boolean;
  dificultadFinal: string;
  cambiosDificultad: number;
  nivelMCERCertificado: boolean;
  informePedagogico?: InformePedagogico;
};

function nombreObjetivo(objetivo: string): string {
  return OBJETIVOS[objetivo] ?? objetivo.replace(/-/g, " ");
}

async function leerJSON(respuesta: Response) {
  const datos = await respuesta.json();

  if (!respuesta.ok) {
    throw new Error(
      typeof datos.error === "string"
        ? datos.error
        : "No se pudo completar la operación."
    );
  }

  return datos;
}

function Acordeon({
  titulo,
  subtitulo,
  abierto,
  alPulsar,
  children,
}: {
  titulo: string;
  subtitulo: string;
  abierto: boolean;
  alPulsar: () => void;
  children: React.ReactNode;
}) {
  return (
    <section className={
      "overflow-hidden rounded-2xl border bg-white shadow-sm transition-colors " +
      (abierto ? "border-indigo-400" : "border-slate-200")
    }>
      <button
        type="button"
        aria-expanded={abierto}
        onClick={alPulsar}
        className="flex w-full items-center justify-between gap-4 p-5 text-left hover:bg-indigo-50/50"
      >
        <span>
          <span className="block font-semibold text-slate-900">
            {titulo}
          </span>
          <span className="mt-1 block text-xs text-slate-600">
            {subtitulo}
          </span>
        </span>
        <span className="text-xl text-indigo-700">
          {abierto ? "−" : "+"}
        </span>
      </button>

      {abierto && (
        <div className="border-t border-slate-200 p-5">
          {children}
        </div>
      )}
    </section>
  );
}

export default function DiagnosticoAdaptativoPage() {
  const [nivel, setNivel] = useState<Nivel>("B1");
  const [perfil, setPerfil] = useState<PerfilProvisional | null>(null);
  const [sesion, setSesion] = useState<Sesion | null>(null);
  const [resultado, setResultado] = useState<Resultado | null>(null);
  const [respuesta, setRespuesta] = useState<number | null>(null);
  const [cargando, setCargando] = useState(false);
  const [error, setError] = useState("");
  const [mostrarAuditoria, setMostrarAuditoria] = useState(false);
  const [seccionAbierta, setSeccionAbierta] =
    useState<Seccion | null>("competencias");

  useEffect(() => {
    const guardado = leerPerfilProvisional();
    if (guardado) {
      setPerfil(guardado);
      setNivel(guardado.nivelDeclarado);
    }
  }, []);

  const pregunta = sesion?.pregunta ?? null;
  const informe = resultado?.informePedagogico;

  const progreso = sesion
    ? Math.min(
        100,
        (sesion.preguntasRespondidas / sesion.maxPreguntas) * 100
      )
    : 0;

  function alternarSeccion(seccion: Seccion) {
    setSeccionAbierta((actual) =>
      actual === seccion ? null : seccion
    );
  }

  async function iniciar() {
    if (cargando) return;
    if (perfil && perfil.idiomaAprendizaje !== "en") {
      setError("El diagnóstico está disponible únicamente para inglés.");
      return;
    }

    setCargando(true);
    setError("");
    setSesion(null);
    setResultado(null);
    setRespuesta(null);
    setMostrarAuditoria(false);
    setSeccionAbierta("competencias");

    try {
      const respuestaAPI = await fetch(API, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        cache: "no-store",
        body: JSON.stringify({
          accion: "iniciar",
          idioma: "en",
          nivel,
        }),
      });

      const datos = await leerJSON(respuestaAPI);
      setSesion(datos.sesion as Sesion);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Error al iniciar la evaluación."
      );
    } finally {
      setCargando(false);
    }
  }

  async function obtenerResultado(sesionId: string) {
    const respuestaAPI = await fetch(
      API +
        "?sesionId=" +
        encodeURIComponent(sesionId) +
        "&tipo=resultado",
      { method: "GET", cache: "no-store" }
    );

    const datos = await leerJSON(respuestaAPI);
    setResultado(datos.resultado as Resultado);
  }

  async function enviarRespuesta() {
    if (
      cargando ||
      !sesion ||
      !pregunta ||
      respuesta === null
    ) {
      return;
    }

    setCargando(true);
    setError("");

    try {
      const respuestaAPI = await fetch(API, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        cache: "no-store",
        body: JSON.stringify({
          accion: "responder",
          sesionId: sesion.sesionId,
          preguntaId: pregunta.id,
          respuesta,
        }),
      });

      const datos = await leerJSON(respuestaAPI);
      const nuevaSesion = datos.sesion as Sesion;

      setSesion(nuevaSesion);
      setRespuesta(null);

      if (nuevaSesion.estado !== "en_curso") {
        await obtenerResultado(nuevaSesion.sesionId);
      }
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Error al procesar la respuesta."
      );
    } finally {
      setCargando(false);
    }
  }

  async function recuperarResultado() {
    if (!sesion || cargando) return;

    setCargando(true);
    setError("");

    try {
      await obtenerResultado(sesion.sesionId);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Error al recuperar el resultado."
      );
    } finally {
      setCargando(false);
    }
  }

  function reiniciar() {
    setSesion(null);
    setResultado(null);
    setRespuesta(null);
    setError("");
    setMostrarAuditoria(false);
    setSeccionAbierta("competencias");
  }

  const competenciasOrdenadas = informe
    ? [...informe.competencias].sort(
        (a, b) => b.porcentaje - a.porcentaje
      )
    : [];

  const fortalezas = competenciasOrdenadas.filter(
    (item) => item.total > 0 && item.aciertos > 0
  );

  const competenciasRefuerzo = competenciasOrdenadas
    .filter((item) => item.errores > 0)
    .sort((a, b) => a.porcentaje - b.porcentaje);

  return (
    <main className="min-h-screen bg-[#F8FAFC] text-slate-900">
      <div className="mx-auto min-h-screen max-w-6xl px-5 py-6 sm:px-8 lg:px-12">
        <header className="flex items-center justify-between gap-4 border-b border-slate-200 pb-5">
          <Link href="/" className="text-2xl font-black tracking-tight" aria-label="Interestalk, inicio">
            INTERES<span className="text-indigo-700">TALK</span>
          </Link>
          <span className="rounded-full border border-indigo-100 bg-indigo-50 px-4 py-2 text-xs font-semibold text-indigo-700">
            PLACEMENT TEST
          </span>
        </header>

        <div className="mx-auto flex min-h-[70vh] max-w-3xl items-center justify-center py-10 sm:py-14">
          {!sesion && !resultado && (
            <section className="w-full text-center">
              <div className="mx-auto mb-8 flex h-20 w-20 items-center justify-center rounded-3xl border border-indigo-100 bg-indigo-50 text-4xl text-indigo-700">
                ✦
              </div>

              <span className="inline-flex rounded-full border border-indigo-100 bg-indigo-50 px-4 py-2 text-sm font-semibold text-indigo-700">
                TU PUNTO DE PARTIDA
              </span>

              <h1 className="mt-6 text-4xl font-bold tracking-tight lg:text-6xl">
                Descubramos
                <br />
                tu nivel.
              </h1>

              <p className="mx-auto mt-6 max-w-xl text-lg leading-relaxed text-slate-600">
                Una evaluación que adapta sus preguntas a tus respuestas
                para conocer mejor tus conocimientos.
              </p>

              {perfil && (
                <div className="mt-7 rounded-2xl border border-teal-200 bg-teal-50 p-5 text-left text-sm text-teal-950">
                  <p className="font-semibold">Preferencias recuperadas</p>
                  <p className="mt-2">Nivel declarado: {perfil.nivelDeclarado} · Objetivo: {perfil.objetivo}</p>
                  <p className="mt-1">Intereses: {perfil.intereses.map((i) => i.nombre).join(", ")}</p>
                  <p className="mt-2 text-xs text-teal-800">Tu nivel sigue siendo provisional y se evaluará con las preguntas del diagnóstico.</p>
                </div>
              )}

              <div className="mt-10 rounded-2xl border border-slate-200 bg-white shadow-sm p-6 text-left">
                <label
                  htmlFor="nivel-inicial"
                  className="mb-3 block text-sm font-medium text-slate-700"
                >
                  ¿Cuál crees que es tu nivel actual?
                </label>

                <select
                  id="nivel-inicial"
                  value={nivel}
                  onChange={(e) => setNivel(e.target.value as Nivel)}
                  className="w-full rounded-xl border border-slate-300 bg-white px-5 py-4 text-slate-900 outline-none focus:border-indigo-600"
                >
                  {NIVELES.map((item) => (
                    <option key={item} value={item}>
                      {item}
                    </option>
                  ))}
                </select>

                <p className="mt-4 text-sm text-slate-500">
                  Esta selección es provisional. La evaluación comenzará
                  cerca del nivel que indiques.
                </p>
              </div>

              <button
                type="button"
                disabled={cargando}
                onClick={iniciar}
                className="mt-8 w-full rounded-2xl bg-indigo-700 px-6 py-4 font-semibold text-white transition-colors hover:bg-indigo-800 disabled:opacity-50 sm:w-auto sm:min-w-72"
              >
                {cargando ? "Preparando..." : "Comenzar evaluación →"}
              </button>

              <p className="mt-6 text-xs text-slate-500">
                Hasta 20 preguntas · Evaluación orientativa
              </p>
            </section>
          )}

          {sesion &&
            sesion.estado === "en_curso" &&
            pregunta &&
            !resultado && (
              <section className="w-full">
                <div className="mb-6 flex items-center justify-between gap-4">
                  <span className="text-sm text-slate-600">
                    Pregunta {sesion.preguntasRespondidas + 1} de un máximo de{" "}
                    {sesion.maxPreguntas}
                  </span>

                  <span className="rounded-full border border-indigo-100 bg-indigo-50 px-4 py-2 text-xs font-semibold text-indigo-700">
                    {NOMBRES[pregunta.competencia]}
                  </span>
                </div>

                <div className="h-2 overflow-hidden rounded-full bg-slate-200">
                  <div
                    className="h-full rounded-full bg-teal-500 transition-all duration-500"
                    style={{ width: progreso + "%" }}
                  />
                </div>

                <div className="mt-12">
                  <p className="mb-4 text-sm font-medium text-indigo-700">
                    CHOOSE THE CORRECT ANSWER
                  </p>

                  {pregunta.contexto && (
                    <div className="mb-8 rounded-2xl border border-slate-200 bg-white shadow-sm p-6 leading-relaxed text-slate-700">
                      {pregunta.contexto}
                    </div>
                  )}

                  <h2 className="text-2xl font-bold leading-relaxed lg:text-3xl">
                    {pregunta.enunciado}
                  </h2>

                  <div className="mt-9 grid gap-3">
                    {pregunta.opciones.map((opcion, indice) => {
                      const seleccionada = respuesta === indice;

                      return (
                        <button
                          key={indice}
                          type="button"
                          disabled={cargando}
                          aria-pressed={seleccionada}
                          onClick={() => setRespuesta(indice)}
                          className={
                            "flex items-center gap-4 rounded-2xl border p-5 text-left transition-all " +
                            (seleccionada
                              ? "border-indigo-600 bg-indigo-50 ring-1 ring-indigo-200"
                              : "border-slate-200 bg-white hover:border-indigo-300 hover:bg-indigo-50/40")
                          }
                        >
                          <span
                            className={
                              "flex h-9 w-9 shrink-0 items-center justify-center rounded-xl border text-sm font-bold " +
                              (seleccionada
                                ? "border-indigo-700 bg-indigo-700 text-white"
                                : "border-slate-300 bg-slate-50 text-slate-700")
                            }
                          >
                            {String.fromCharCode(65 + indice)}
                          </span>
                          <span className="font-medium">{opcion}</span>
                        </button>
                      );
                    })}
                  </div>

                  <button
                    type="button"
                    disabled={respuesta === null || cargando}
                    onClick={enviarRespuesta}
                    className="mt-10 w-full rounded-2xl bg-indigo-700 px-8 py-4 font-semibold text-white transition-colors hover:bg-indigo-800 disabled:cursor-not-allowed disabled:opacity-40"
                  >
                    {cargando ? "Procesando..." : "Continuar →"}
                  </button>
                </div>
              </section>
            )}

          {resultado && (
            <section className="w-full text-center">
              <div className="mx-auto mb-7 flex h-20 w-20 items-center justify-center rounded-3xl border border-indigo-100 bg-indigo-50 text-4xl text-indigo-700">
                ✓
              </div>

              <h2 className="text-4xl font-bold">
                Tu punto de partida
              </h2>

              <p className="mt-4 text-slate-600">
                Explora tus resultados a tu ritmo.
              </p>

              <div className="mt-9 grid grid-cols-2 gap-4">
                <div className="rounded-2xl border border-slate-200 bg-white shadow-sm p-6">
                  <p className="text-sm text-slate-600">
                    Respuestas correctas
                  </p>
                  <p className="mt-3 text-4xl font-bold text-indigo-700">
                    {resultado.aciertos}/{resultado.preguntasRespondidas}
                  </p>
                </div>

                <div className="rounded-2xl border border-slate-200 bg-white shadow-sm p-6">
                  <p className="text-sm text-slate-600">
                    Puntuación
                  </p>
                  <p className="mt-3 text-4xl font-bold text-indigo-700">
                    {resultado.porcentaje}%
                  </p>
                </div>
              </div>

              {resultado.estado === "incompleto" && (
                <div className="mt-6 rounded-2xl border border-amber-200 bg-amber-50 p-5 text-sm text-amber-900">
                  La evaluación no ha alcanzado toda la cobertura prevista.
                  Los resultados son parciales.
                </div>
              )}

              {informe ? (
                <div className="mt-9 space-y-3 text-left">
                  <h3 className="mb-5 text-lg font-semibold">
                    Explora tus resultados
                  </h3>

                  <Acordeon
                    titulo="Tus competencias"
                    subtitulo="Grammar, Vocabulary y Reading"
                    abierto={seccionAbierta === "competencias"}
                    alPulsar={() => alternarSeccion("competencias")}
                  >
                    <div className="space-y-6">
                      {informe.competencias.map((item) => (
                        <div key={item.competencia}>
                          <div className="mb-2 flex items-center justify-between">
                            <span>{NOMBRES[item.competencia]}</span>
                            <span className="text-sm text-slate-600">
                              {item.aciertos}/{item.total} · {item.porcentaje}%
                            </span>
                          </div>
                          <div className="h-2 overflow-hidden rounded-full bg-slate-200">
                            <div
                              className="h-full rounded-full bg-teal-500"
                              style={{ width: item.porcentaje + "%" }}
                            />
                          </div>
                        </div>
                      ))}
                    </div>
                  </Acordeon>

                  <Acordeon
                    titulo="Dificultad de las preguntas"
                    subtitulo="Resultados por nivel de los ejercicios"
                    abierto={seccionAbierta === "niveles"}
                    alPulsar={() => alternarSeccion("niveles")}
                  >
                    <p className="mb-5 text-sm text-slate-600">
                      Estos niveles indican la dificultad de las preguntas,
                      no una certificación MCER.
                    </p>

                    <div className="space-y-3">
                      {informe.niveles
                        .filter((item) => item.total > 0)
                        .map((item) => (
                          <div
                            key={item.nivel}
                            className="flex justify-between border-b border-slate-200 pb-3"
                          >
                            <span className="font-semibold text-indigo-700">
                              {item.nivel}
                            </span>
                            <span className="text-sm text-slate-700">
                              {item.aciertos}/{item.total} · {item.porcentaje}%
                            </span>
                          </div>
                        ))}
                    </div>
                  </Acordeon>

                  <Acordeon
                    titulo="Tus fortalezas"
                    subtitulo="Competencias con aciertos observados"
                    abierto={seccionAbierta === "fortalezas"}
                    alPulsar={() => alternarSeccion("fortalezas")}
                  >
                    {fortalezas.length > 0 ? (
                      <div className="space-y-4">
                        {fortalezas.map((item) => (
                          <div
                            key={item.competencia}
                            className="rounded-xl border border-slate-200 bg-slate-50 p-4"
                          >
                            <p className="font-semibold">
                              {NOMBRES[item.competencia]}
                            </p>
                            <p className="mt-2 text-sm text-slate-600">
                              {item.aciertos} respuestas correctas de{" "}
                              {item.total} ({item.porcentaje}%).
                              Es un resultado observado que seguiremos
                              comprobando con nuevas actividades.
                            </p>
                          </div>
                        ))}
                      </div>
                    ) : (
                      <p className="text-sm text-slate-600">
                        Todavía necesitamos más evidencias para identificar
                        fortalezas.
                      </p>
                    )}
                  </Acordeon>

                  <Acordeon
                    titulo="Tus próximos objetivos"
                    subtitulo="Contenidos que conviene reforzar"
                    abierto={seccionAbierta === "refuerzo"}
                    alPulsar={() => alternarSeccion("refuerzo")}
                  >
                    {informe.recomendaciones.length > 0 ? (
                      <div className="space-y-3">
                        {informe.recomendaciones.map((item) => (
                          <div
                            key={item.prioridad + "-" + item.objetivo}
                            className="rounded-xl border border-slate-200 bg-slate-50 p-4"
                          >
                            <p className="font-semibold">
                              {item.prioridad}.{" "}
                              {nombreObjetivo(item.objetivo)}
                            </p>
                            <p className="mt-1 text-xs text-indigo-700">
                              {NOMBRES[item.competencia]} · {item.nivel}
                            </p>
                            <p className="mt-2 text-sm text-slate-600">
                              {item.motivo}
                            </p>
                          </div>
                        ))}
                      </div>
                    ) : (
                      <p className="text-sm text-slate-600">
                        No se han registrado errores en los objetivos
                        evaluados. Seguiremos recogiendo evidencias.
                      </p>
                    )}
                  </Acordeon>

                  <Acordeon
                    titulo="Tu plan de aprendizaje"
                    subtitulo="Primeras actividades recomendadas"
                    abierto={seccionAbierta === "plan"}
                    alPulsar={() => alternarSeccion("plan")}
                  >
                    <p className="mb-4 text-sm text-slate-600">
                      Esta propuesta inicial se basa en los resultados
                      observados y se revisará con nuevas actividades.
                    </p>

                    {competenciasRefuerzo.length > 0 ? (
                      <div className="space-y-4">
                        {competenciasRefuerzo.map((item, indice) => (
                          <div
                            key={item.competencia}
                            className="rounded-xl border border-slate-200 bg-slate-50 p-4"
                          >
                            <p className="font-semibold text-indigo-700">
                              Paso {indice + 1} · {NOMBRES[item.competencia]}
                            </p>
                            <p className="mt-2 text-sm text-slate-700">
                              Practicar contenidos de esta competencia
                              y comprobar el progreso con nuevas preguntas.
                            </p>
                            <p className="mt-2 text-xs text-slate-500">
                              Resultado observado: {item.aciertos}/
                              {item.total} · {item.porcentaje}%
                            </p>
                          </div>
                        ))}
                      </div>
                    ) : (
                      <p className="text-sm text-slate-600">
                        Continuaremos con actividades de consolidación
                        y nuevas comprobaciones de aprendizaje.
                      </p>
                    )}
                  </Acordeon>
                </div>
              ) : (
                <div className="mt-7 rounded-2xl border border-amber-200 bg-amber-50 p-5 text-sm text-amber-900">
                  El informe pedagógico no está disponible en esta sesión.
                </div>
              )}

              <p className="mt-8 text-sm text-slate-500">
                Evaluación orientativa. No determina ni certifica un nivel MCER.
              </p>

              {DESARROLLO && (
                <div className="mt-8 rounded-2xl border border-indigo-200 bg-white shadow-sm p-5 text-left">
                  <button
                    type="button"
                    aria-expanded={mostrarAuditoria}
                    onClick={() => setMostrarAuditoria((actual) => !actual)}
                    className="flex w-full items-center justify-between text-sm font-semibold text-indigo-700"
                  >
                    <span>Auditoría técnica · Solo desarrollo</span>
                    <span>{mostrarAuditoria ? "−" : "+"}</span>
                  </button>

                  {mostrarAuditoria && (
                    <div className="mt-5 space-y-3 border-t border-slate-200 pt-5 text-sm">
                      <div>
                        <p className="text-xs text-slate-500">ID de sesión</p>
                        <p className="mt-1 break-all font-mono text-xs text-slate-700">
                          {resultado.sesionId}
                        </p>
                      </div>

                      {[
                        ["Estado", resultado.estado],
                        [
                          "Motivo de finalización",
                          resultado.motivoFinalizacion ?? "No especificado",
                        ],
                        ["Dificultad final", resultado.dificultadFinal],
                        [
                          "Cambios de dificultad",
                          String(resultado.cambiosDificultad),
                        ],
                        [
                          "Cobertura mínima",
                          resultado.coberturaCompleta
                            ? "Alcanzada"
                            : "No alcanzada",
                        ],
                        [
                          "Preguntas respondidas",
                          resultado.preguntasRespondidas + "/20",
                        ],
                        [
                          "Informe pedagógico",
                          informe ? "Disponible" : "No disponible",
                        ],
                        [
                          "Recomendaciones",
                          String(informe?.recomendaciones.length ?? 0),
                        ],
                      ].map(([etiqueta, valor]) => (
                        <div
                          key={etiqueta}
                          className="flex justify-between gap-4 border-b border-slate-200 pb-2"
                        >
                          <span className="text-slate-600">{etiqueta}</span>
                          <span className="text-right font-medium">{valor}</span>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              )}

              <button
                type="button"
                onClick={reiniciar}
                className="mt-8 rounded-2xl bg-indigo-700 px-8 py-4 font-semibold text-white hover:bg-indigo-800"
              >
                Repetir evaluación
              </button>
            </section>
          )}

          {sesion &&
            sesion.estado !== "en_curso" &&
            !resultado && (
              <section className="w-full text-center">
                <h2 className="text-3xl font-bold">
                  Evaluación finalizada
                </h2>
                <p className="mt-4 text-slate-600">
                  Estamos recuperando tu resultado.
                </p>
                <button
                  type="button"
                  disabled={cargando}
                  onClick={recuperarResultado}
                  className="mt-8 rounded-2xl bg-indigo-700 px-6 py-4 font-semibold text-white disabled:opacity-50"
                >
                  {cargando ? "Recuperando..." : "Ver resultado"}
                </button>
              </section>
            )}
        </div>

        {error && (
          <div
            role="alert"
            className="fixed bottom-6 left-1/2 z-50 w-[calc(100%-2rem)] max-w-lg -translate-x-1/2 rounded-2xl border border-red-200 bg-red-50 p-5 text-center text-sm text-red-800 shadow-xl"
          >
            <p>{error}</p>
            <button
              type="button"
              onClick={() => setError("")}
              className="mt-3 rounded-xl border border-red-300 px-4 py-2 hover:bg-red-100"
            >
              Cerrar
            </button>
          </div>
        )}

        <footer className="border-t border-slate-200 py-6 text-center text-xs text-slate-500">
          INTERESTALK · Learn what you love. · Evaluación orientativa
        </footer>
      </div>
    </main>
  );
}
