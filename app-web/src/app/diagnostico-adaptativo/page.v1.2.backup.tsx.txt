"use client";

import { useState } from "react";

// ======================================================
// INTERESTALK - DIAGNOSTICO ADAPTATIVO V1.2

// ======================================================

type Competencia = "grammar" | "vocabulary" | "reading";
type Nivel = "A1" | "A2" | "B1" | "B2" | "C1" | "C2";

type Pregunta = {
  id: string;
  idioma: string;
  nivel: string;
  competencia: Competencia;
  objetivo: string;
  enunciado: string;
  contexto?: string;
  opciones: string[];
};

type Sesion = {
  sesionId: string;
  estado: "en_curso" | "completado" | "incompleto";
  pregunta: Pregunta | null;
  preguntasRespondidas: number;
  maxPreguntas: number;
  expiraEn: string;
};

type ResultadoCompetencia = {
  competencia: Competencia;
  total: number;
  aciertos: number;
  errores: number;
  porcentaje: number;
};

type ResultadoNivel = {
  nivel: Nivel;
  total: number;
  aciertos: number;
  errores: number;
  porcentaje: number;
};

type Recomendacion = {
  prioridad: number;
  competencia: Competencia;
  objetivo: string;
  nivel: Nivel;
  motivo: string;
};

type InformePedagogico = {
  total: number;
  aciertos: number;
  errores: number;
  porcentaje: number;
  competencias: ResultadoCompetencia[];
  niveles: ResultadoNivel[];
  recomendaciones: Recomendacion[];
  nivelMCERCertificado: false;
};

type Resultado = {
  sesionId: string;
  estado: "completado" | "incompleto";
  motivoFinalizacion: string | null;
  preguntasRespondidas: number;
  aciertos: number;
  porcentaje: number;
  conteos: Record<Competencia, number>;
  coberturaCompleta: boolean;
  dificultadFinal: string;
  cambiosDificultad: number;
  nivelMCERCertificado: boolean;
  informePedagogico?: InformePedagogico;
};

const API = "/api/diagnostico-adaptativo";
const DESARROLLO = process.env.NODE_ENV === "development";

const NIVELES: Nivel[] = ["A1", "A2", "B1", "B2", "C1", "C2"];

const COMPETENCIAS: Competencia[] = [
  "grammar",
  "vocabulary",
  "reading",
];

const NOMBRES: Record<Competencia, string> = {
  grammar: "Grammar",
  vocabulary: "Vocabulary",
  reading: "Reading",
};

  "second-conditional": "Segundo condicional",

  "relative-clauses": "Oraciones de relativo",

  inversion: "Estructuras de inversión",

function nombreObjetivo(objetivo: string): string {
  return OBJETIVOS[objetivo] ??
    objetivo.replace(/-/g, " ");
}

async function leerJSON(respuesta: Response) {
  const datos = await respuesta.json();

  if (!respuesta.ok) {
    throw new Error(
      typeof datos.error === "string"
        ? datos.error
        : "No se pudo completar la operación."
    );
  }

  return datos;
}

export default function DiagnosticoAdaptativoPage() {
  const [nivel, setNivel] = useState<Nivel>("B1");
  const [sesion, setSesion] = useState<Sesion | null>(null);
  const [resultado, setResultado] = useState<Resultado | null>(null);
  const [respuesta, setRespuesta] = useState<number | null>(null);
  const [cargando, setCargando] = useState(false);
  const [error, setError] = useState("");
  const [mostrarAuditoria, setMostrarAuditoria] = useState(false);

  const pregunta = sesion?.pregunta ?? null;
  const informe = resultado?.informePedagogico;

  const progreso = sesion
    ? Math.min(
        100,
        (sesion.preguntasRespondidas / sesion.maxPreguntas) * 100
      )
    : 0;

  async function iniciar() {
    if (cargando) return;

    setCargando(true);
    setError("");
    setSesion(null);
    setResultado(null);
    setRespuesta(null);
    setMostrarAuditoria(false);

    try {
      const respuestaAPI = await fetch(API, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        cache: "no-store",
        body: JSON.stringify({
          accion: "iniciar",
          idioma: "en",
          nivel,
        }),
      });

      const datos = await leerJSON(respuestaAPI);
      setSesion(datos.sesion as Sesion);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Error al iniciar la evaluación."
      );
    } finally {
      setCargando(false);
    }
  }

  async function obtenerResultado(sesionId: string) {
    const respuestaAPI = await fetch(
      API +
        "?sesionId=" +
        encodeURIComponent(sesionId) +
        "&tipo=resultado",
      { method: "GET", cache: "no-store" }
    );

    const datos = await leerJSON(respuestaAPI);
    setResultado(datos.resultado as Resultado);
  }

  async function enviarRespuesta() {
    if (
      cargando ||
      !sesion ||
      !pregunta ||
      respuesta === null
    ) {
      return;
    }

    setCargando(true);
    setError("");

    try {
      const respuestaAPI = await fetch(API, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        cache: "no-store",
        body: JSON.stringify({
          accion: "responder",
          sesionId: sesion.sesionId,
          preguntaId: pregunta.id,
          respuesta,
        }),
      });

      const datos = await leerJSON(respuestaAPI);
      const nuevaSesion = datos.sesion as Sesion;

      setSesion(nuevaSesion);
      setRespuesta(null);

      if (nuevaSesion.estado !== "en_curso") {
        await obtenerResultado(nuevaSesion.sesionId);
      }
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Error al procesar la respuesta."
      );
    } finally {
      setCargando(false);
    }
  }

  async function recuperarResultado() {
    if (!sesion || cargando) return;

    setCargando(true);
    setError("");

    try {
      await obtenerResultado(sesion.sesionId);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Error al recuperar el resultado."
      );
    } finally {
      setCargando(false);
    }
  }

  function reiniciar() {
    setSesion(null);
    setResultado(null);
    setRespuesta(null);
    setError("");
    setMostrarAuditoria(false);
  }

  return (
    <main className="min-h-screen bg-[#090b12] text-white">
      <div className="mx-auto min-h-screen max-w-6xl px-6 py-8 lg:px-12">
        <header className="flex items-center justify-between border-b border-white/10 pb-6">
          <div className="text-2xl font-black tracking-tight">
            INTERES<span className="text-violet-400">TALK</span>
          </div>
          <span className="rounded-full border border-white/10 bg-white/5 px-4 py-2 text-xs text-gray-300">
            PLACEMENT TEST
          </span>
        </header>

        <div className="mx-auto flex min-h-[75vh] max-w-3xl items-center justify-center py-12">
          {!sesion && !resultado && (
            <section className="w-full text-center">
              <div className="mx-auto mb-8 flex h-20 w-20 items-center justify-center rounded-3xl border border-violet-400/30 bg-violet-500/10 text-4xl text-violet-300">
                ✦
              </div>

              <span className="inline-flex rounded-full border border-violet-400/30 bg-violet-500/10 px-4 py-2 text-sm text-violet-300">
                TU PUNTO DE PARTIDA
              </span>

              <h1 className="mt-6 text-4xl font-bold tracking-tight lg:text-6xl">
                Descubramos
                <br />
                tu nivel.
              </h1>

              <p className="mx-auto mt-6 max-w-xl text-lg leading-relaxed text-gray-400">
                Una evaluación que adapta sus preguntas a tus respuestas
                para conocer mejor tus conocimientos.
              </p>

              <div className="mt-10 rounded-2xl border border-white/10 bg-white/5 p-6 text-left">
                <label
                  htmlFor="nivel-inicial"
                  className="mb-3 block text-sm font-medium text-gray-300"
                >
                  ¿Cuál crees que es tu nivel actual?
                </label>

                <select
                  id="nivel-inicial"
                  value={nivel}
                  onChange={(e) => setNivel(e.target.value as Nivel)}
                  className="w-full rounded-xl border border-white/10 bg-[#151823] px-5 py-4 text-white outline-none focus:border-violet-500"
                >
                  {NIVELES.map((item) => (
                    <option key={item} value={item}>
                      {item}
                    </option>
                  ))}
                </select>

                <p className="mt-4 text-sm text-gray-500">
                  Esta selección es provisional. La evaluación comenzará
                  cerca del nivel que indiques.
                </p>
              </div>

              <button
                type="button"
                disabled={cargando}
                onClick={iniciar}
                className="mt-8 w-full rounded-2xl bg-violet-600 px-6 py-4 font-semibold transition-colors hover:bg-violet-500 disabled:opacity-50 sm:w-auto sm:min-w-72"
              >
                {cargando ? "Preparando..." : "Comenzar evaluación →"}
              </button>

              <p className="mt-6 text-xs text-gray-500">
                Hasta 20 preguntas · Evaluación orientativa
              </p>
            </section>
          )}

          {sesion &&
            sesion.estado === "en_curso" &&
            pregunta &&
            !resultado && (
              <section className="w-full">
                <div className="mb-6 flex items-center justify-between gap-4">
                  <span className="text-sm text-gray-400">
                    Pregunta {sesion.preguntasRespondidas + 1} de un máximo de{" "}
                    {sesion.maxPreguntas}
                  </span>

                  <span className="rounded-full border border-violet-400/20 bg-violet-500/10 px-4 py-2 text-xs font-medium text-violet-300">
                    {NOMBRES[pregunta.competencia]}
                  </span>
                </div>

                <div className="h-2 overflow-hidden rounded-full bg-white/10">
                  <div
                    className="h-full rounded-full bg-violet-500 transition-all duration-500"
                    style={{ width: progreso + "%" }}
                  />
                </div>

                <div className="mt-12">
                  <p className="mb-4 text-sm font-medium text-violet-400">
                    CHOOSE THE CORRECT ANSWER
                  </p>

                  {pregunta.contexto && (
                    <div className="mb-8 rounded-2xl border border-white/10 bg-white/5 p-6 leading-relaxed text-gray-300">
                      {pregunta.contexto}
                    </div>
                  )}

                  <h2 className="text-2xl font-bold leading-relaxed lg:text-3xl">
                    {pregunta.enunciado}
                  </h2>

                  <div className="mt-9 grid gap-3">
                    {pregunta.opciones.map((opcion, indice) => {
                      const seleccionada = respuesta === indice;

                      return (
                        <button
                          key={indice}
                          type="button"
                          disabled={cargando}
                          aria-pressed={seleccionada}
                          onClick={() => setRespuesta(indice)}
                          className={
                            "flex items-center gap-4 rounded-2xl border p-5 text-left transition-all " +
                            (seleccionada
                              ? "border-violet-400 bg-violet-600/20"
                              : "border-white/10 bg-white/5 hover:border-violet-400/40 hover:bg-white/10")
                          }
                        >
                          <span
                            className={
                              "flex h-9 w-9 shrink-0 items-center justify-center rounded-xl border text-sm font-bold " +
                              (seleccionada
                                ? "border-violet-400 bg-violet-600"
                                : "border-white/20 bg-white/5")
                            }
                          >
                            {String.fromCharCode(65 + indice)}
                          </span>
                          <span className="font-medium">{opcion}</span>
                        </button>
                      );
                    })}
                  </div>

                  <button
                    type="button"
                    disabled={respuesta === null || cargando}
                    onClick={enviarRespuesta}
                    className="mt-10 w-full rounded-2xl bg-violet-600 px-8 py-4 font-semibold transition-colors hover:bg-violet-500 disabled:cursor-not-allowed disabled:opacity-40"
                  >
                    {cargando ? "Procesando..." : "Continuar →"}
                  </button>
                </div>
              </section>
            )}

          {resultado && (
            <section className="w-full text-center">
              <div className="mx-auto mb-7 flex h-20 w-20 items-center justify-center rounded-3xl border border-violet-400/30 bg-violet-500/10 text-4xl text-violet-300">
                ✓
              </div>

              <h2 className="text-4xl font-bold">Tu punto de partida</h2>
              <p className="mt-4 text-gray-400">
                Estos resultados nos ayudarán a personalizar tu aprendizaje.
              </p>

              <div className="mt-9 grid grid-cols-2 gap-4">
                <div className="rounded-2xl border border-white/10 bg-white/5 p-6">
                  <p className="text-sm text-gray-400">
                    Respuestas correctas
                  </p>
                  <p className="mt-3 text-4xl font-bold text-violet-300">
                    {resultado.aciertos}/{resultado.preguntasRespondidas}
                  </p>
                </div>

                <div className="rounded-2xl border border-white/10 bg-white/5 p-6">
                  <p className="text-sm text-gray-400">Puntuación</p>
                  <p className="mt-3 text-4xl font-bold text-violet-300">
                    {resultado.porcentaje}%
                  </p>
                </div>
              </div>

              {informe ? (
                <>
                  <div className="mt-7 rounded-2xl border border-white/10 bg-white/5 p-6 text-left">
                    <h3 className="text-lg font-semibold">
                      Tus competencias
                    </h3>

                    <div className="mt-6 space-y-6">
                      {informe.competencias.map((item) => (
                        <div key={item.competencia}>
                          <div className="mb-2 flex items-center justify-between gap-4">
                            <span className="font-medium text-gray-200">
                              {NOMBRES[item.competencia]}
                            </span>
                            <span className="text-sm text-gray-400">
                              {item.aciertos}/{item.total} · {item.porcentaje}%
                            </span>
                          </div>

                          <div className="h-2 overflow-hidden rounded-full bg-white/10">
                            <div
                              className="h-full rounded-full bg-violet-500"
                              style={{
                                width: item.porcentaje + "%",
                              }}
                            />
                          </div>
                        </div>
                      ))}
                    </div>
                  </div>

                  <div className="mt-7 rounded-2xl border border-white/10 bg-white/5 p-6 text-left">
                    <h3 className="text-lg font-semibold">
                      Dificultad de las preguntas
                    </h3>

                    <p className="mt-2 text-sm text-gray-400">
                      Resultados según la dificultad de los ejercicios,
                      no una certificación de nivel.
                    </p>

                    <div className="mt-5 space-y-3">
                      {informe.niveles
                        .filter((item) => item.total > 0)
                        .map((item) => (
                          <div
                            key={item.nivel}
                            className="flex items-center justify-between border-b border-white/10 pb-3 last:border-0"
                          >
                            <span className="font-medium text-violet-300">
                              {item.nivel}
                            </span>
                            <span className="text-sm text-gray-300">
                              {item.aciertos}/{item.total} · {item.porcentaje}%
                            </span>
                          </div>
                        ))}
                    </div>
                  </div>

                  <div className="mt-7 rounded-2xl border border-white/10 bg-white/5 p-6 text-left">
                    <h3 className="text-lg font-semibold">
                      Tus próximos objetivos
                    </h3>

                    <p className="mt-2 text-sm text-gray-400">
                      Hemos identificado contenidos que conviene practicar
                      y comprobar con nuevas actividades.
                    </p>

                    {informe.recomendaciones.length > 0 ? (
                      <div className="mt-6 space-y-3">
                        {informe.recomendaciones.map((item) => (
                          <div
                            key={
                              item.prioridad +
                              "-" +
                              item.objetivo
                            }
                            className="rounded-xl border border-white/10 bg-[#10121c] p-4"
                          >
                            <div className="flex items-start gap-3">
                              <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-violet-500/15 text-sm font-bold text-violet-300">
                                {item.prioridad}
                              </span>

                              <div>
                                <p className="font-semibold text-white">
                                  {nombreObjetivo(item.objetivo)}
                                </p>
                                <p className="mt-1 text-xs text-violet-300">
                                  {NOMBRES[item.competencia]} · {item.nivel}
                                </p>
                                <p className="mt-2 text-sm leading-relaxed text-gray-400">
                                  {item.motivo}
                                </p>
                              </div>
                            </div>
                          </div>
                        ))}
                      </div>
                    ) : (
                      <p className="mt-5 text-sm text-gray-400">
                        No se han registrado errores en los objetivos evaluados.
                        Seguiremos recogiendo evidencias en las próximas actividades.
                      </p>
                    )}
                  </div>
                </>
              ) : (
                <div className="mt-7 rounded-2xl border border-amber-500/30 bg-amber-500/10 p-5 text-sm text-amber-200">
                  El informe pedagógico no está disponible en esta sesión.
                  El resultado general sí se ha recuperado.
                </div>
              )}

              {resultado.estado === "incompleto" && (
                <div className="mt-6 rounded-2xl border border-amber-500/30 bg-amber-500/10 p-5 text-sm text-amber-200">
                  La evaluación no ha alcanzado toda la cobertura prevista.
                  Estos resultados son parciales.
                </div>
              )}

              <p className="mt-7 text-sm leading-relaxed text-gray-500">
                Evaluación orientativa basada en las respuestas observadas.
                No determina ni certifica un nivel MCER.
              </p>

              {DESARROLLO && (
                <div className="mt-8 rounded-2xl border border-violet-400/20 bg-[#121320] p-5 text-left">
                  <button
                    type="button"
                    aria-expanded={mostrarAuditoria}
                    onClick={() =>
                      setMostrarAuditoria((actual) => !actual)
                    }
                    className="flex w-full items-center justify-between gap-3 text-sm font-semibold text-violet-300"
                  >
                    <span>Auditoría técnica · Solo desarrollo</span>
                    <span>{mostrarAuditoria ? "−" : "+"}</span>
                  </button>

                  {mostrarAuditoria && (
                    <div className="mt-5 space-y-3 border-t border-white/10 pt-5 text-sm">
                      <div>
                        <p className="text-xs text-gray-500">ID de sesión</p>
                        <p className="mt-1 break-all font-mono text-xs text-gray-300">
                          {resultado.sesionId}
                        </p>
                      </div>

                      {[
                        ["Estado", resultado.estado],
                        [
                          "Motivo de finalización",
                          resultado.motivoFinalizacion ?? "No especificado",
                        ],
                        ["Dificultad final", resultado.dificultadFinal],
                        [
                          "Cambios de dificultad",
                          String(resultado.cambiosDificultad),
                        ],
                        [
                          "Cobertura mínima",
                          resultado.coberturaCompleta
                            ? "Alcanzada"
                            : "No alcanzada",
                        ],
                        [
                          "Preguntas respondidas",
                          resultado.preguntasRespondidas +
                            "/20",
                        ],
                        [
                          "Informe pedagógico",
                          informe ? "Disponible" : "No disponible",
                        ],
                        [
                          "Recomendaciones",
                          String(informe?.recomendaciones.length ?? 0),
                        ],
                      ].map(([etiqueta, valor]) => (
                        <div
                          key={etiqueta}
                          className="flex items-start justify-between gap-4 border-b border-white/10 pb-2"
                        >
                          <span className="text-gray-400">{etiqueta}</span>
                          <span className="text-right font-medium text-white">
                            {valor}
                          </span>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              )}

              <button
                type="button"
                onClick={reiniciar}
                className="mt-8 rounded-2xl bg-violet-600 px-8 py-4 font-semibold hover:bg-violet-500"
              >
                Repetir evaluación
              </button>
            </section>
          )}

          {sesion &&
            sesion.estado !== "en_curso" &&
            !resultado && (
              <section className="w-full text-center">
                <h2 className="text-3xl font-bold">
                  Evaluación finalizada
                </h2>
                <p className="mt-4 text-gray-400">
                  Estamos recuperando tu resultado.
                </p>
                <button
                  type="button"
                  disabled={cargando}
                  onClick={recuperarResultado}
                  className="mt-8 rounded-2xl bg-violet-600 px-6 py-4 font-semibold disabled:opacity-50"
                >
                  {cargando ? "Recuperando..." : "Ver resultado"}
                </button>
              </section>
            )}
        </div>

        {error && (
          <div
            role="alert"
            className="fixed bottom-6 left-1/2 z-50 w-[calc(100%-2rem)] max-w-lg -translate-x-1/2 rounded-2xl border border-red-500/30 bg-[#29151c] p-5 text-center text-sm text-red-200 shadow-xl"
          >
            <p>{error}</p>
            <button
              type="button"
              onClick={() => setError("")}
              className="mt-3 rounded-xl border border-red-400/30 px-4 py-2 hover:bg-white/10"
            >
              Cerrar
            </button>
          </div>
        )}

        <footer className="border-t border-white/10 py-6 text-center text-xs text-gray-500">
          INTERESTALK · Learn what you love. · Evaluación orientativa
        </footer>
      </div>
    </main>
  );
}
